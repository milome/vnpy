# 多周期K线叠加实现方案

## 概述

本文档详细描述如何实现多周期K线在同一个图表上的叠加显示，所有周期统一使用1分钟时间轴作为基准。

## 核心设计原则

1. **统一时间轴**：所有周期K线都映射到1分钟时间轴上
2. **时间对齐**：大周期K线的起始和结束时间对应到1分钟K线的索引位置
3. **实时同步**：1分钟K线更新时，自动触发大周期K线的更新
4. **性能优化**：只绘制可见范围内的K线，使用缓存减少重复计算

## 数据结构设计

### MultiTimeframeBarManager

```python
from datetime import datetime
from typing import Optional, Tuple, Dict, List
from vnpy.trader.object import BarData

class MultiTimeframeBarManager:
    """
    多周期K线管理器
    
    核心功能：
    1. 管理1分钟K线作为时间轴基准
    2. 管理大周期K线数据
    3. 建立大周期K线到1分钟时间轴的映射
    """
    
    def __init__(self):
        # ========== 1分钟时间轴基准 ==========
        # 1分钟K线数据（按时间排序）
        self._base_bars: List[BarData] = []
        
        # datetime -> index 映射（快速查找）
        self._datetime_index_map: Dict[datetime, int] = {}
        
        # index -> datetime 映射（反向查找）
        self._index_datetime_map: Dict[int, datetime] = {}
        
        # ========== 大周期K线数据 ==========
        # 格式：timeframe -> List[BarData]
        # 注意：使用List而不是Dict，因为需要保持顺序
        self._timeframe_bars: Dict[str, List[BarData]] = {
            "5m": [],
            "1h": [],
            "4h": [],
            "1d": []
        }
        
        # ========== 时间范围映射 ==========
        # 格式：timeframe -> {bar_datetime: (start_index, end_index)}
        # start_index: 大周期K线起始时间对应的1分钟索引
        # end_index: 大周期K线结束时间对应的1分钟索引（不包含）
        self._timeframe_range_map: Dict[str, Dict[datetime, Tuple[int, int]]] = {
            "5m": {},
            "1h": {},
            "4h": {},
            "1d": {}
        }
        
        # ========== 缓存 ==========
        # 缓存周期结束时间计算（避免重复计算）
        self._period_end_cache: Dict[Tuple[str, datetime], datetime] = {}
        
    # ========== 1分钟时间轴管理 ==========
    
    def set_base_timeline(self, one_minute_bars: List[BarData]) -> None:
        """
        设置1分钟K线作为时间轴基准
        
        Args:
            one_minute_bars: 1分钟K线数据列表（必须已按时间排序）
        """
        self._base_bars = one_minute_bars.copy()
        self._datetime_index_map.clear()
        self._index_datetime_map.clear()
        
        # 建立双向映射
        for index, bar in enumerate(self._base_bars):
            dt = bar.datetime
            self._datetime_index_map[dt] = index
            self._index_datetime_map[index] = dt
        
        # 清空缓存（因为基准时间轴改变了）
        self._period_end_cache.clear()
        
        # 重新计算所有大周期K线的时间范围映射
        self._rebuild_timeframe_mappings()
    
    def add_base_bar(self, bar: BarData) -> int:
        """
        添加单个1分钟K线（用于实时更新）
        
        Args:
            bar: 1分钟K线数据
            
        Returns:
            新添加的K线的索引位置
        """
        dt = bar.datetime
        
        # 检查是否已存在
        if dt in self._datetime_index_map:
            # 更新现有K线
            index = self._datetime_index_map[dt]
            self._base_bars[index] = bar
            return index
        
        # 添加到末尾
        index = len(self._base_bars)
        self._base_bars.append(bar)
        self._datetime_index_map[dt] = index
        self._index_datetime_map[index] = dt
        
        # 触发大周期K线重新计算（如果这个新K线影响了某个大周期）
        self._update_affected_timeframes(dt)
        
        return index
    
    def get_base_index(self, dt: datetime) -> Optional[int]:
        """
        获取指定时间对应的1分钟索引
        
        Args:
            dt: 时间
            
        Returns:
            索引位置，如果不存在则返回None
        """
        return self._datetime_index_map.get(dt)
    
    def get_base_datetime(self, index: int) -> Optional[datetime]:
        """
        获取指定索引对应的1分钟时间
        
        Args:
            index: 索引位置
            
        Returns:
            时间，如果不存在则返回None
        """
        return self._index_datetime_map.get(index)
    
    def get_base_count(self) -> int:
        """获取1分钟K线总数"""
        return len(self._base_bars)
    
    # ========== 大周期K线管理 ==========
    
    def add_timeframe_bars(
        self, 
        timeframe: str, 
        bars: List[BarData]
    ) -> None:
        """
        添加大周期K线数据并建立时间轴映射
        
        Args:
            timeframe: 周期标识（"5m", "1h", "4h", "1d"）
            bars: 大周期K线数据列表（必须已按时间排序）
        """
        if timeframe not in self._timeframe_bars:
            self._timeframe_bars[timeframe] = []
            self._timeframe_range_map[timeframe] = {}
        
        # 添加K线数据
        self._timeframe_bars[timeframe] = bars.copy()
        
        # 建立时间范围映射
        self._build_timeframe_mapping(timeframe)
    
    def get_timeframe_bars(self, timeframe: str) -> List[BarData]:
        """
        获取指定周期的所有K线
        
        Args:
            timeframe: 周期标识
            
        Returns:
            K线数据列表
        """
        return self._timeframe_bars.get(timeframe, [])
    
    def get_timeframe_range(
        self, 
        timeframe: str, 
        bar_datetime: datetime
    ) -> Optional[Tuple[int, int]]:
        """
        获取大周期K线的时间范围（索引范围）
        
        Args:
            timeframe: 周期标识
            bar_datetime: 大周期K线的开始时间
            
        Returns:
            (start_index, end_index) 或 None
        """
        range_map = self._timeframe_range_map.get(timeframe, {})
        return range_map.get(bar_datetime)
    
    def update_timeframe_bar(
        self, 
        timeframe: str, 
        bar: BarData
    ) -> None:
        """
        更新大周期K线（用于实时更新）
        
        Args:
            timeframe: 周期标识
            bar: 实时K线数据（可能未完成）
        """
        if timeframe not in self._timeframe_bars:
            self._timeframe_bars[timeframe] = []
            self._timeframe_range_map[timeframe] = {}
        
        dt = bar.datetime
        
        # 查找是否已存在
        bars = self._timeframe_bars[timeframe]
        existing_index = None
        for i, existing_bar in enumerate(bars):
            if existing_bar.datetime == dt:
                existing_index = i
                break
        
        if existing_index is not None:
            # 更新现有K线
            bars[existing_index] = bar
        else:
            # 添加新K线（实时K线）
            bars.append(bar)
            # 保持排序
            bars.sort(key=lambda b: b.datetime)
        
        # 重新计算时间范围映射
        self._build_timeframe_mapping(timeframe)
    
    # ========== 时间范围映射计算 ==========
    
    def _rebuild_timeframe_mappings(self) -> None:
        """重新计算所有大周期K线的时间范围映射"""
        for timeframe in self._timeframe_bars.keys():
            self._build_timeframe_mapping(timeframe)
    
    def _build_timeframe_mapping(self, timeframe: str) -> None:
        """
        建立指定周期的时间范围映射
        
        Args:
            timeframe: 周期标识
        """
        if timeframe not in self._timeframe_bars:
            return
        
        bars = self._timeframe_bars[timeframe]
        range_map = self._timeframe_range_map[timeframe]
        range_map.clear()
        
        for bar in bars:
            dt = bar.datetime
            start_ix, end_ix = self._calculate_timeframe_range(timeframe, dt)
            if start_ix is not None and end_ix is not None:
                range_map[dt] = (start_ix, end_ix)
    
    def _calculate_timeframe_range(
        self, 
        timeframe: str, 
        start_datetime: datetime
    ) -> Tuple[Optional[int], Optional[int]]:
        """
        计算大周期K线的时间范围（索引范围）
        
        Args:
            timeframe: 周期标识
            start_datetime: 大周期K线的开始时间
            
        Returns:
            (start_index, end_index) 或 (None, None)
        """
        # 1. 获取起始索引
        start_ix = self.get_base_index(start_datetime)
        if start_ix is None:
            return (None, None)
        
        # 2. 计算结束时间
        end_datetime = self._calculate_period_end(timeframe, start_datetime)
        if end_datetime is None:
            return (None, None)
        
        # 3. 获取结束索引
        # 注意：end_datetime 是下一个周期的开始时间，所以应该查找前一个1分钟K线
        # 或者查找最接近的1分钟K线
        end_ix = self._find_closest_base_index(end_datetime, before=True)
        if end_ix is None:
            # 如果找不到，使用当前最新的索引（实时K线的情况）
            end_ix = len(self._base_bars) - 1
        
        return (start_ix, end_ix)
    
    def _calculate_period_end(
        self, 
        timeframe: str, 
        start_datetime: datetime
    ) -> Optional[datetime]:
        """
        计算大周期K线的结束时间（下一个周期的开始时间）
        
        Args:
            timeframe: 周期标识
            start_datetime: 周期开始时间
            
        Returns:
            周期结束时间（下一个周期的开始时间），如果无法计算则返回None
        """
        # 检查缓存
        cache_key = (timeframe, start_datetime)
        if cache_key in self._period_end_cache:
            return self._period_end_cache[cache_key]
        
        from datetime import timedelta
        
        if timeframe == "5m":
            # 5分钟：简单的时间增量
            end_datetime = start_datetime + timedelta(minutes=5)
        elif timeframe == "1h":
            # 1小时：简单的时间增量
            end_datetime = start_datetime + timedelta(hours=1)
        elif timeframe == "4h":
            # 4小时：需要特殊处理（HKFE交易时段）
            end_datetime = self._calculate_4h_period_end(start_datetime)
        elif timeframe == "1d":
            # 日线：下一天的同一时间
            end_datetime = start_datetime + timedelta(days=1)
        else:
            return None
        
        # 缓存结果
        self._period_end_cache[cache_key] = end_datetime
        return end_datetime
    
    def _calculate_4h_period_end(self, start_datetime: datetime) -> datetime:
        """
        计算4小时周期的结束时间（HKFE交易时段）
        
        4小时周期规则：
        1. 17:15-21:14 (第一根4H)
        2. 21:15-01:14 (第二根4H，跨午夜)
        3. 01:15-11:29 (第三根4H，跨休市：01:15-03:00 + 09:15-11:29)
        4. 11:30-16:29 (第四根4H，跨午休：11:30-12:00 + 13:00-16:29)
        
        Args:
            start_datetime: 4小时周期开始时间
            
        Returns:
            4小时周期结束时间（下一个周期的开始时间）
        """
        from datetime import datetime, timedelta
        
        hour = start_datetime.hour
        minute = start_datetime.minute
        
        # 根据开始时间判断是哪个4小时周期
        if hour == 17 and minute == 15:
            # 第一根：17:15 -> 21:15
            end_datetime = start_datetime.replace(hour=21, minute=15)
        elif hour == 21 and minute == 15:
            # 第二根：21:15 -> 次日01:15
            end_datetime = start_datetime + timedelta(days=1)
            end_datetime = end_datetime.replace(hour=1, minute=15)
        elif hour == 1 and minute == 15:
            # 第三根：01:15 -> 当日11:30
            end_datetime = start_datetime.replace(hour=11, minute=30)
        elif hour == 11 and minute == 30:
            # 第四根：11:30 -> 次日17:15
            end_datetime = start_datetime + timedelta(days=1)
            end_datetime = end_datetime.replace(hour=17, minute=15)
        else:
            # 默认：4小时后
            end_datetime = start_datetime + timedelta(hours=4)
        
        return end_datetime
    
    def _find_closest_base_index(
        self, 
        target_datetime: datetime, 
        before: bool = True
    ) -> Optional[int]:
        """
        查找最接近目标时间的1分钟索引
        
        Args:
            target_datetime: 目标时间
            before: 如果为True，查找目标时间之前的最近索引；如果为False，查找之后的最近索引
            
        Returns:
            索引位置，如果找不到则返回None
        """
        # 精确匹配
        exact_index = self.get_base_index(target_datetime)
        if exact_index is not None:
            return exact_index
        
        # 二分查找最接近的索引
        if not self._base_bars:
            return None
        
        # 使用二分查找
        left = 0
        right = len(self._base_bars) - 1
        closest_index = None
        
        while left <= right:
            mid = (left + right) // 2
            mid_datetime = self._base_bars[mid].datetime
            
            if mid_datetime == target_datetime:
                return mid
            elif mid_datetime < target_datetime:
                closest_index = mid
                left = mid + 1
            else:
                right = mid - 1
        
        if before:
            return closest_index
        else:
            # 查找之后的最近索引
            if closest_index is not None and closest_index < len(self._base_bars) - 1:
                return closest_index + 1
            return None
    
    def _update_affected_timeframes(self, new_base_datetime: datetime) -> None:
        """
        更新受影响的大周期K线（当新增1分钟K线时）
        
        Args:
            new_base_datetime: 新添加的1分钟K线时间
        """
        # 检查每个大周期，看是否有K线需要更新
        for timeframe in self._timeframe_bars.keys():
            bars = self._timeframe_bars[timeframe]
            
            # 查找包含这个时间的K线
            for bar in bars:
                range_map = self._timeframe_range_map[timeframe]
                start_ix, end_ix = range_map.get(bar.datetime, (None, None))
                
                if start_ix is not None and end_ix is not None:
                    # 检查新时间是否在这个范围内
                    new_ix = self.get_base_index(new_base_datetime)
                    if new_ix is not None and start_ix <= new_ix <= end_ix:
                        # 需要更新end_ix（实时K线的情况）
                        # 注意：这里不直接更新，而是标记需要重新计算
                        pass
            
            # 重新计算时间范围映射（简化处理）
            self._build_timeframe_mapping(timeframe)
```

## 绘制实现

### MultiTimeframeCandleItem

```python
from vnpy.chart.item import ChartItem
from vnpy.trader.object import BarData
from vnpy.trader.ui import QtGui, QtCore
import pyqtgraph as pg

class MultiTimeframeCandleItem(ChartItem):
    """
    支持多周期K线绘制的CandleItem
    
    特点：
    1. 支持跨索引位置绘制（大周期K线）
    2. 自动处理不同周期的K线宽度
    3. 优化绘制性能（只绘制可见范围）
    """
    
    def __init__(self, manager: MultiTimeframeBarManager, timeframe: str = "1m"):
        """
        Args:
            manager: 多周期K线管理器
            timeframe: 当前绘制的周期标识
        """
        super().__init__(manager)
        self._timeframe = timeframe
        self._manager = manager
    
    def _draw_bar_picture(
        self, 
        ix: int, 
        bar: BarData
    ) -> QtGui.QPicture:
        """
        绘制K线（支持跨索引位置）
        
        Args:
            ix: 起始索引位置
            bar: K线数据
        """
        # 1. 获取时间范围
        if self._timeframe == "1m":
            # 1分钟K线：占据一个索引位置
            start_ix = ix
            end_ix = ix
            center_ix = float(ix)
        else:
            # 大周期K线：获取时间范围
            range_map = self._manager.get_timeframe_range(self._timeframe, bar.datetime)
            if range_map is None:
                # 如果无法映射，使用默认值
                start_ix = ix
                end_ix = ix
                center_ix = float(ix)
            else:
                start_ix, end_ix = range_map
                center_ix = (start_ix + end_ix) / 2.0
        
        # 2. 计算K线宽度
        width = end_ix - start_ix + 1
        
        # 3. 绘制K线
        candle_picture = QtGui.QPicture()
        painter = QtGui.QPainter(candle_picture)
        
        # 设置颜色
        if bar.close_price >= bar.open_price:
            painter.setPen(self._up_pen)
            painter.setBrush(self._black_brush)
        else:
            painter.setPen(self._down_pen)
            painter.setBrush(self._down_brush)
        
        # 绘制影线（在中心位置）
        if bar.high_price > bar.low_price:
            painter.drawLine(
                QtCore.QPointF(center_ix, bar.high_price),
                QtCore.QPointF(center_ix, bar.low_price)
            )
        
        # 绘制实体
        if bar.open_price == bar.close_price:
            # 十字星：绘制水平线，跨越时间范围
            painter.drawLine(
                QtCore.QPointF(start_ix, bar.open_price),
                QtCore.QPointF(end_ix + 1, bar.open_price)
            )
        else:
            # 实体：绘制矩形，宽度为时间范围
            from vnpy.chart.base import BAR_WIDTH
            rect = QtCore.QRectF(
                start_ix - BAR_WIDTH,
                min(bar.open_price, bar.close_price),
                width + BAR_WIDTH * 2,
                abs(bar.close_price - bar.open_price)
            )
            painter.drawRect(rect)
        
        painter.end()
        return candle_picture
```

```python
class ChartWidget:
    """支持多周期K线叠加的ChartWidget"""
    
    def __init__(self, parent=None):
        super().__init__(parent)
        
        # 使用多周期K线管理器
        self._manager = MultiTimeframeBarManager()
        
        # 存储不同周期的绘制项
        self._timeframe_items: Dict[str, MultiTimeframeCandleItem] = {}
        
        # ... 其他初始化代码 ...
    
    def add_timeframe(
        self, 
        timeframe: str, 
        plot_name: str = "candle"
    ) -> None:
        """
        添加一个周期的K线绘制
        
        Args:
            timeframe: 周期标识（"1m", "5m", "1h", "4h", "1d"）
            plot_name: 绘制的plot名称
        """
        # 创建绘制项
        item = MultiTimeframeCandleItem(self._manager, timeframe)
        self._timeframe_items[timeframe] = item
        
        # 添加到plot
        plot = self._plots.get(plot_name)
        if plot:
            plot.addItem(item)
    
    def load_multi_timeframe_data(
        self, 
        one_minute_bars: List[BarData],
        timeframe_bars: Dict[str, List[BarData]]
    ) -> None:
        """
        加载多周期K线数据
        
        Args:
            one_minute_bars: 1分钟K线数据
            timeframe_bars: 大周期K线数据 {timeframe: [BarData]}
        """
        # 1. 设置1分钟时间轴基准
        self._manager.set_base_timeline(one_minute_bars)
        
        # 2. 添加大周期K线
        for timeframe, bars in timeframe_bars.items():
            self._manager.add_timeframe_bars(timeframe, bars)
        
        # 3. 更新所有绘制项
        for item in self._timeframe_items.values():
            item.update_history([])  # 触发重新绘制
    
    def update_realtime_bar(
        self, 
        timeframe: str, 
        bar: BarData
    ) -> None:
        """
        更新实时K线
        
        Args:
            timeframe: 周期标识
            bar: 实时K线数据
        """
        if timeframe == "1m":
            # 1分钟K线：添加到基准时间轴
            self._manager.add_base_bar(bar)
        else:
            # 大周期K线：更新对应周期
            self._manager.update_timeframe_bar(timeframe, bar)
        
        # 更新绘制项
        item = self._timeframe_items.get(timeframe)
        if item:
            item.update_bar(bar)
```

## 性能优化建议

1. **只绘制可见范围**：已实现，通过 `paint()` 方法的 `exposedRect` 参数
2. **缓存时间范围映射**：避免重复计算
3. **延迟更新**：对于实时K线，可以批量更新而不是每次tick都更新
4. **使用索引查找**：使用 `datetime -> index` 映射，避免线性搜索
